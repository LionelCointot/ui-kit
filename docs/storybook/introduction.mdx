<Meta title="Introduction" />

export const Go = ({ to, className, children }) => {
  const context = useContext(DocsContext);
  return (
    <a
      className={className}
      href={`./?path=${to}`}
      target="_top"
      onClick={(event) => {
        if (event.button !== 0 || event.metaKey || event.ctrlKey || event.shiftKey || event.altKey) return;
        event.preventDefault();
        context.channel.emit('navigateUrl', `?path=${to}`);
      }}
    >
      {children}
    </a>
  );
};

## How to read this site

- Each component has an **Overview** page: what it is for, a **Playground** whose controls change the live component, the generated API table (inputs and outputs), then focused examples.
- Use the **Theme** button in the toolbar to switch any example between the light and dark themes.
- The **Code** panel under each example shows the Angular template to copy.
- Demo content is in French, the library's default language; see [Getting started](?path=/docs/getting-started--overview) to switch labels to English.
